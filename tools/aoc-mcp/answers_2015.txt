#!/bin/bash
# Submit all verified 2015 answers via aoc-mcp; stops at the first wrong one.
# Correct/WrongLevel continue; Incorrect/TooHigh/TooLow/errors abort.
cd "$(dirname "$0")/.."
BIN=./zig-out/bin/aoc-mcp
LOG=/tmp/aoc_submit_2015.log

submit() { # day part answer
  local day=$1 part=$2 answer=$3
  local out
  out=$("$BIN" submit 2015 "$day" "$part" "$answer" 2>&1 | tail -1)
  echo "$(date '+%H:%M:%S') 2015/$day/$part $answer -> $out" >> "$LOG"
  case "$out" in
    Correct|WrongLevel) return 0 ;;
    *) echo "$(date '+%H:%M:%S') ABORT: $out" >> "$LOG"; return 1 ;;
  esac
}
